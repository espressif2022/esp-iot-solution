# Compile independent plugin sets and both P4 SRM macro-block sizes.
# Disabled plugins have no mock
# driver symbols, so accidental dependencies fail the link (including CPU-only).
foreach(variant cpu ppa dma2d both ppa_mb16 both_color both_legacy both_legacy_ecc both_legacy_flash)
    set(has_ppa 0)
    set(has_dma2d 0)
    if(variant STREQUAL "ppa" OR variant MATCHES "^both" OR variant STREQUAL "ppa_mb16")
        set(has_ppa 1)
    endif()
    if(variant STREQUAL "dma2d" OR variant MATCHES "^both")
        set(has_dma2d 1)
    endif()
    set(ppa_mb_height 32)
    if(variant STREQUAL "ppa_mb16")
        set(ppa_mb_height 16)
    endif()
    add_executable(test_executor_${variant} test_executor.c
        ${PRESENT_OS_SOURCES} ${PRESENT_PIXEL_SOURCES})
    target_include_directories(test_executor_${variant} PRIVATE ../../src ../../private_include ../../include support)
    target_compile_definitions(test_executor_${variant} PRIVATE PRESENT_HOST=1
        PRESENT_HOST_PPA=${has_ppa} PRESENT_HOST_DMA2D=${has_dma2d}
        PRESENT_HOST_PPA_MB_HEIGHT=${ppa_mb_height})
    if(variant MATCHES "^both_(color|legacy)")
        target_compile_definitions(test_executor_${variant} PRIVATE PRESENT_IDF_COLOR_CONVERT_DMA2D_API=1)
    endif()
    if(variant MATCHES "^both_legacy")
        target_compile_definitions(test_executor_${variant} PRIVATE PRESENT_IDF_DMA2D_ALIGNMENT_API=0)
    else()
        target_compile_definitions(test_executor_${variant} PRIVATE PRESENT_IDF_DMA2D_ALIGNMENT_API=1)
    endif()
    if(variant STREQUAL "both_legacy_ecc")
        target_compile_definitions(test_executor_${variant} PRIVATE CONFIG_SPIRAM_ECC_ENABLE=1)
    endif()
    if(variant STREQUAL "both_legacy_flash")
        target_compile_definitions(test_executor_${variant} PRIVATE PRESENT_IDF_EFUSE_ENCRYPTION_API=0)
    endif()
    target_compile_options(test_executor_${variant} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    target_link_libraries(test_executor_${variant} PRIVATE Threads::Threads)
    if(PRESENT_SANITIZE)
        target_compile_options(test_executor_${variant} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
        target_link_options(test_executor_${variant} PRIVATE -fsanitize=address,undefined -no-pie)
    endif()
    add_test(NAME executor_${variant} COMMAND test_executor_${variant})
    set_tests_properties(executor_${variant} PROPERTIES TIMEOUT 15)
endforeach()
