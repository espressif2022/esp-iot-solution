set(PRESENT_SOURCES src/present_os.c src/present_event.c src/presenter.c src/pixel/pixel_engine.c src/backend/gram.c
    src/platform/pixel_accel.c src/platform/lcd_spi.c src/platform/lcd_scanout.c src/backend/te_composer.c src/backend/te.c src/backend/fb.c)

if(COMMAND idf_component_register)
    idf_component_register(SRCS ${PRESENT_SOURCES} INCLUDE_DIRS include
        PRIV_INCLUDE_DIRS src REQUIRES esp_common esp_lcd
        PRIV_REQUIRES freertos esp_timer heap esp_mm esp_hw_support esp_driver_ppa esp_driver_dma)
    target_compile_features(${COMPONENT_LIB} PRIVATE c_std_11)
    return()
endif()

cmake_minimum_required(VERSION 3.16)
project(esp_display_present_v2 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(PRESENT_SANITIZE "Enable address and undefined-behavior sanitizers" OFF)
find_package(Threads REQUIRED)
add_library(present_v2 STATIC ${PRESENT_SOURCES})
target_include_directories(present_v2 PUBLIC include tests/support PRIVATE src)
target_compile_definitions(present_v2 PRIVATE PV2_HOST=1)
target_compile_options(present_v2 PRIVATE -Wall -Wextra -Werror)
target_link_libraries(present_v2 PUBLIC Threads::Threads)
if(PRESENT_SANITIZE)
    target_compile_options(present_v2 PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer -fno-pie)
    target_link_options(present_v2 PUBLIC -fsanitize=address,undefined -no-pie)
endif()
enable_testing()
foreach(name records coverage repair accel scanout rotation pixel gram lcd_spi presenter te_composer te presenter_te fb event scheduler)
    add_executable(test_${name} tests/test_${name}.c)
    target_link_libraries(test_${name} PRIVATE present_v2)
    target_compile_definitions(test_${name} PRIVATE PV2_HOST=1)
    target_compile_options(test_${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME ${name} COMMAND test_${name})
    set_tests_properties(${name} PROPERTIES TIMEOUT 15)
endforeach()
target_link_options(test_rotation PRIVATE -Wl,--wrap=pv2_alloc_pixels -Wl,--wrap=pv2_free)
add_executable(header_smoke tests/header_smoke.cpp)
target_link_libraries(header_smoke PRIVATE present_v2)
target_compile_options(header_smoke PRIVATE -Wall -Wextra -Werror)
add_test(NAME headers_cpp COMMAND header_smoke)
