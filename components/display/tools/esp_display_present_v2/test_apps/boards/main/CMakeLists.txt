set(APP_SOURCE main.c)
if(IDF_TARGET STREQUAL "esp32s3")
    set(BOARD_SOURCE board_s3.c)
    set(BOARD_DEPS esp_lcd_gc9503 esp_lcd_panel_io_additions esp_io_expander_tca9554)
elseif(IDF_TARGET STREQUAL "esp32p4")
    set(BOARD_SOURCE board_p4.c executor_check.c)
    set(BOARD_DEPS esp_lcd_ek79007 esp_hw_support)
elseif(IDF_TARGET STREQUAL "esp32s31")
    set(APP_SOURCE qspi_main.c)
    set(BOARD_SOURCE board_qspi.c)
    set(BOARD_DEPS esp_lcd_co5300 esp_driver_spi efuse)
else()
    message(FATAL_ERROR "Only the confirmed S3/P4 boards are supported")
endif()
idf_component_register(SRCS ${APP_SOURCE} ${BOARD_SOURCE} INCLUDE_DIRS "."
    REQUIRES esp_display_present_v2 esp_display_present esp_lcd esp_timer esp_psram esp_driver_gpio esp_driver_i2c ${BOARD_DEPS})

if(IDF_TARGET STREQUAL "esp32p4")
    target_compile_definitions(${COMPONENT_LIB} PRIVATE PV2_BOARD_EXECUTOR_CHECK=1)
endif()
